02_run_pipeline_all_samples.sh — Full pipeline (loop ทุก sample)

Pipeline หลักที่รัน FastQC → HISAT2 → SAMtools → StringTie ครบในสคริปต์เดียว

#!/bin/bash
ml load FastQC
ml load hisat2
ml load samtools
ml load stringtie

raw_dir="/fs4/Mining_data/Method/RNAseq/Oui/20260428_HN00271762_TRR_Report/rawdata/"
work_dir="/fs4/Mining_data/Method/RNAseq/Oui/"
ref_index="/fs4/Mining_data/Method/RNAseq/Oui/Reference/IRGSP-1.0_genome"
gtf="/fs4/Mining_data/Method/RNAseq/Oui/Reference/IRGSP-1.0_representative_transcript_exon_2026-02-05.gtf"
sample_list="${work_dir}scripts/sample_list.txt"

while IFS= read -r SAMPLE || <a href="#"> -n "${SAMPLE}" </a>; do
    <a href="#">| "${SAMPLE}" =~ ^# </a> && continue

    LOG="${work_dir}logs/${SAMPLE}_pipeline.log"
    R1="${raw_dir}${SAMPLE}_1.fastq.gz"
    R2="${raw_dir}${SAMPLE}_2.fastq.gz"

    # Step 1: FastQC
    mkdir -p "${work_dir}FastQC/${SAMPLE}"
    fastqc -o "${work_dir}FastQC/${SAMPLE}" --threads 8 "${R1}" "${R2}"

    # Step 2: HISAT2
    hisat2 -p 32 --dta -x "${ref_index}" \
        -1 "${R1}" -2 "${R2}" \
        -S "${work_dir}Mapping/${SAMPLE}.sam"

    # Step 3: SAMtools sort
    samtools sort -@ 8 -o "${work_dir}Mapping/${SAMPLE}.bam" \
        "${work_dir}Mapping/${SAMPLE}.sam"
    rm -f "${work_dir}Mapping/${SAMPLE}.sam"

    # Step 4: StringTie
    mkdir -p "${work_dir}Transcript/${SAMPLE}"
    stringtie -e -B -p 32 -G "${gtf}" \
        -o "${work_dir}Transcript/${SAMPLE}/${SAMPLE}.gtf" \
        "${work_dir}Mapping/${SAMPLE}.bam"

done < "${sample_list}" 2>&1 | tee -a "${LOG}"

Features:
- รัน samples ทั้งหมดอัตโนมัติจาก sample_list.txt
- บันทึก log แยกทีละ sample ที่ logs/<SAMPLE>_pipeline.log
- ใช้ --resume flag เพื่อข้าม sample ที่มี BAM อยู่แล้ว
- แสดง pass/fail summary เมื่อรันครบทุก sample
