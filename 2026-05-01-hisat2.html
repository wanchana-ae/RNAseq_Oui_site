💻 Pipeline Script

# Step 2: HISAT2
hisat2 -p 32 --dta \
    -x /fs4/Mining_data/Method/RNAseq/Oui/Reference/IRGSP-1.0_genome \
    -1 ${raw_dir}${SAMPLE}_1.fastq.gz \
    -2 ${raw_dir}${SAMPLE}_2.fastq.gz \
    -S ${work_dir}Mapping/${SAMPLE}.sam

# Step 3: SAMtools sort
samtools sort -@ 8 -o ${work_dir}Mapping/${SAMPLE}.bam \
    ${work_dir}Mapping/${SAMPLE}.sam
rm -f ${work_dir}Mapping/${SAMPLE}.sam

# Step 4: StringTie
stringtie -e -B -p 32 \
    -G ${gtf} \
    -o ${work_dir}Transcript/${SAMPLE}/${SAMPLE}.gtf \
    ${work_dir}Mapping/${SAMPLE}.bam
